```swift
import Foundation
public import SwiftUI

// The account's My Tags as the site's My Tags page (`/mytags`) shows them, read live on both hosts on 2026-11-22
// (port of PR 462 item 10). The app keeps one cached copy, every tag set with its tags, and draws every tag's My Tags
// symbol from it (design approved 2026-11-22).

/// A colour as the site writes it, `#RRGGBB`, kept as its three 8-bit channels.
///
/// An edit sends the colour back in the same form, so the channels are stored as the site has them rather than as a
/// `Color`, whose floating-point components would not give the exact value back.
public struct SiteColor: Codable, Equatable, Hashable, Sendable {
    public let red: Int
    public let green: Int
    public let blue: Int

    /// Each channel is clamped to 0...255.
    public init(red: Int, green: Int, blue: Int) {
        self.red = min(max(red, 0), 255)
        self.green = min(max(green, 0), 255)
        self.blue = min(max(blue, 0), 255)
    }

    /// Reads `#RRGGBB` or `RRGGBB` in either case; anything else is nil.
    public init?(hex: String) {
        let digits = hex.hasPrefix("#") ? hex.dropFirst() : Substring(hex)
        guard digits.count == 6, digits.allSatisfy(\.isHexDigit), let value = Int(digits, radix: 16)
        else { return nil }
        self.init(red: (value >> 16) & 0xFF, green: (value >> 8) & 0xFF, blue: value & 0xFF)
    }

    /// The site colour nearest to `color` in sRGB; a channel outside sRGB is clamped.
    public init(_ color: Color) {
        let resolved = color.resolve(in: EnvironmentValues())
        self.init(
            red: Int((Double(resolved.red) * 255).rounded()),
            green: Int((Double(resolved.green) * 255).rounded()),
            blue: Int((Double(resolved.blue) * 255).rounded())
        )
    }

    /// `#RRGGBB` in upper case, the form an edit sends and the page shows.
    public var hex: String {
        String(format: "#%02lX%02lX%02lX", red, green, blue)
    }

    public var color: Color {
        Color(.sRGB, red: Double(red) / 255, green: Double(green) / 255, blue: Double(blue) / 255)
    }
}

/// How a tag is kept in My Tags, the one choice the site's page offers per tag: neither watched nor hidden (the
/// design's Normal), watched or hidden.
public enum MyTagKind: CaseIterable, Equatable, Hashable, Sendable {
    case flagged
    case watched
    case hidden
}

/// One tag of a tag set, as its row on the site's My Tags page shows it.
public struct MyTag: Codable, Equatable, Hashable, Identifiable, Sendable {
    /// The weights the site's edit accepts ("Weight must be between 99 and -99").
    public static let weightRange = -99...99
    /// The weight a new tag starts at, the site's add row's default.
    public static let defaultWeight = 10

    /// The site's tag id, the one the gallery page's tag menu uses too.
    public let id: Int
    /// The full `namespace:tag`, as the row's preview titles it.
    public let name: String
    public var isWatched: Bool
    public var isHidden: Bool
    /// The tag's own colour; nil takes its set's.
    public var color: SiteColor?
    public var weight: Int

    public init(id: Int, name: String, isWatched: Bool, isHidden: Bool, color: SiteColor?, weight: Int) {
        self.id = id
        self.name = name
        self.isWatched = isWatched
        self.isHidden = isHidden
        self.color = color
        self.weight = weight
    }

    /// The two flags as the one choice the site's page makes; Hidden wins if both are set, and setting a kind sets
    /// both.
    public var kind: MyTagKind {
        get {
            if isHidden { return .hidden }
            return isWatched ? .watched : .flagged
        }
        set {
            isWatched = newValue == .watched
            isHidden = newValue == .hidden
        }
    }
}

/// A tag to add to a tag set, as the site's add row takes it.
public struct MyTagDraft: Equatable, Sendable {
    /// The full `namespace:tag`, or a bare tag, which the site files under temp.
    public let name: String
    public let kind: MyTagKind
    public let color: SiteColor?
    public let weight: Int

    public init(name: String, kind: MyTagKind, color: SiteColor?, weight: Int) {
        self.name = name
        self.kind = kind
        self.color = color
        self.weight = weight
    }
}

/// One tag set of My Tags: its name, whether it is enabled, its default colour, and its tags in the page's order.
public struct MyTagSet: Codable, Equatable, Hashable, Identifiable, Sendable {
    public let id: Int
    public let name: String
    /// Only the tags of an enabled set take effect, on the site and in the app's highlight.
    public var isEnabled: Bool
    /// The colour of the set's tags that have none of their own.
    public var color: SiteColor?
    public var tags: [MyTag]

    public init(id: Int, name: String, isEnabled: Bool, color: SiteColor?, tags: [MyTag]) {
        self.id = id
        self.name = name
        self.isEnabled = isEnabled
        self.color = color
        self.tags = tags
    }

/// How a tag of an enabled tag set stands out wherever tags are drawn: its state's symbol, in its colour (its own,

/// The account's My Tags: every tag set, in the order the site's picker lists them, and the set last chosen on the

        self.selectedTagSetID = selectedTagSetID

    public func applying(_ page: MyTagsPage) -> MyTags {
        let tagSets = page.tagSetIDs.compactMap { id in
            id == page.tagSet.id ? page.tagSet : self.tagSets.first(where: { $0.id == id })
        }
        return MyTags(tagSets: tagSets, selectedTagSetID: selectedTagSetID)
    }

    /// The cache with `tagSet` in place of the set of the same id.
    public func updating(_ tagSet: MyTagSet) -> MyTags {
        MyTags(tagSets: tagSets.map({ $0.id == tagSet.id ? tagSet : $0 }), selectedTagSetID: selectedTagSetID)

            guard tagSet.id == tagSetID else { return tagSet }
```

```swift
public import AppModels
import Foundation
public import Kanna

extension Parser {
    /// Reads one tag set's page of the site's My Tags (`/mytags`, read live on both hosts on 2026-11-22).
    ///
    /// The page shows the set its picker has selected; the picker lists every set, each named with its tag count in
    /// parentheses, which is dropped. Each `div#usertag_<id>` after the add row (`usertag_0`) is one tag: its
    /// preview's `title` is the full `namespace:tag` (the preview's text may abbreviate the namespace); Watched and
    /// Hidden are set when their box carries the `checked` attribute (their `onchange` text contains
    /// `checked=false`, so a substring test would be wrong); its colour and weight are its inputs' values. A refused
    /// form post comes back as the same page with the reason in `#msg`. The inline script's `apiuid` and `apikey` are
    /// the edit API's credentials.
    ///
    /// A page without the credentials, the picker, its selected set or the tag list is not a My Tags page, and a row
    /// without its name or weight, or with a colour that is not `#RRGGBB`, fails the page rather than losing the row.
    public static func parseMyTags(doc: any HTMLDocument) throws(AppError) -> MyTagsPage {
        guard let apikey = parseScriptVariable(name: "apikey", doc: doc), !apikey.isEmpty,
              let apiuid = parseAPIUID(doc: doc),
              let picker = doc.at_xpath("//div[@id='tagset_outer']//select"),
              let selected = picker.at_xpath(".//option[@selected]"),
              let tagSetID = selected["value"].flatMap({ Int($0) }),
              let optionText = selected.text?.trimmingCharacters(in: .whitespacesAndNewlines),
              let usertags = doc.at_xpath("//div[@id='usertags_outer']")
        else { throw .parseFailed }

        var tags = [MyTag]()
        for row in usertags.xpath("./div[starts-with(@id, 'usertag_')]") {
            guard let tagID = row["id"].flatMap({ Int($0.dropFirst("usertag_".count)) })
            else { throw .parseFailed }
            guard tagID != 0 else { continue }
            guard let name = row.at_xpath(".//div[@id='tagpreview_\(tagID)']")?["title"], !name.isEmpty,
                  let weight = row.at_xpath(".//input[@id='tagweight_\(tagID)']")?["value"].flatMap({ Int($0) })
            else { throw .parseFailed }
            let color = try siteColor(row.at_xpath(".//input[@id='tagcolor_\(tagID)']")?["value"])
            tags.append(MyTag(
                id: tagID,
                name: name,
                isWatched: row.at_xpath(".//input[@id='tagwatch_\(tagID)']")?["checked"] != nil,
                isHidden: row.at_xpath(".//input[@id='taghide_\(tagID)']")?["checked"] != nil,
                color: color,
                weight: weight
            ))
        }

        let tagSetColor = try siteColor(doc.at_xpath("//input[@id='tagcolor']")?["value"])
        let message = doc.at_xpath("//div[@id='msg']")?.text?.trimmingCharacters(in: .whitespacesAndNewlines) ?? ""
        return MyTagsPage(
            tagSet: MyTagSet(
                id: tagSetID,
                name: optionText.wholeMatch(of: /(.*) \(\d+\)/).map({ String($0.output.1) }) ?? optionText,
                isEnabled: doc.at_xpath("//input[@id='tagset_enable']")?["checked"] != nil,
                color: tagSetColor,
                tags: tags
            ),
            tagSetIDs: picker.xpath(".//option").compactMap({ option in option["value"].flatMap({ Int($0) }) }),
            apiuid: apiuid,
            apikey: apikey,
            message: message.isEmpty ? nil : String(message.prefix(200))
        )
    }

    /// The script's `var apiuid = <int>;`, which unlike the key the site writes unquoted.
    private static func parseAPIUID(doc: any HTMLDocument) -> Int? {
        for script in doc.xpath("//script") {
            if let text = script.text, let match = text.firstMatch(of: /var apiuid\s*=\s*(\d+)\s*;/) {
                return Int(match.output.1)
            }
        }
        return nil
    }

    /// An input's colour: nil when empty (the set's or the site's default applies), else `#RRGGBB`; anything else
    /// fails the page.
    private static func siteColor(_ value: String?) throws(AppError) -> SiteColor? {
        let text = value?.trimmingCharacters(in: .whitespaces) ?? ""
        guard !text.isEmpty else { return nil }
        guard let color = SiteColor(hex: text) else { throw .parseFailed }
        return color
    }
}
```

#### S18: `.capture/fv8/privacy_check.py` (new scratch script, Task 1; not committed)

It reads the two live captures, learns the owner's credentials and full tag names from them, and reports, for each file
it is given, which kind of value appears in it. It prints `HIT <kind> <file>` lines and then `clean` or `<n> hits`, and

```python
import re
import sys
from pathlib import Path

CAPTURES = [Path(".capture/fv8/eh-mytags-raw.html"), Path(".capture/fv8/ex-mytags-raw.html")]


def secrets():
    found = {}
    for capture in CAPTURES:
        text = capture.read_text(encoding="utf-8", errors="replace")
        for match in re.finditer(r"var apiuid\s*=\s*(\d+)\s*;", text):
            found[match.group(1)] = "apiuid"
        for match in re.finditer(r'var apikey\s*=\s*"([0-9a-f]+)"\s*;', text):
            found[match.group(1)] = "apikey"
        for match in re.finditer(r'<div id="tagpreview_\d+"[^>]*\btitle="([^"]+)"', text):
            found[match.group(1)] = "tag"
    kinds = set(found.values())
    assert {"apiuid", "apikey", "tag"} <= kinds, "the captures no longer hold what this check looks for"
    return found


def main(paths):
    found = secrets()
    hits = 0
    for name in paths:
        text = Path(name).read_text(encoding="utf-8", errors="replace")
        for value, kind in sorted(found.items(), key=lambda item: item[1]):
            if re.search(r"(?<![0-9A-Za-z])" + re.escape(value) + r"(?![0-9A-Za-z])", text):
                print("HIT", kind, name)
                hits += 1
    print("clean" if hits == 0 else f"{hits} hits")
    return 1 if hits else 0


if __name__ == "__main__":
    sys.exit(main(sys.argv[1:]))
```

```python
def canonical(catalog_object):
    return json.dumps(catalog_object, indent=2, ensure_ascii=False, separators=(",", " : "))


def unit(value):
    return {"stringUnit": {"state": "translated", "value": value}}


def entry_object(key, values):
    entry = {"extractionState": "manual", "localizations": {locale: unit(values[locale]) for locale in LOCALES}}
    if key in NOT_TRANSLATED:
        entry["shouldTranslate"] = False
    return entry


def entry_lines(form, key, values):
    if form == "compact":
        lines = [f'    "{key}" : {{', '      "extractionState" : "manual",', '      "localizations" : {']
        for index, locale in enumerate(LOCALES):
            value = json.dumps(values[locale], ensure_ascii=False)
            comma = "," if index < len(LOCALES) - 1 else ""
            lines.append(f'        "{locale}" : {{ "stringUnit" : {{ "state" : "translated", "value" : {value} }} }}{comma}')
        return lines + ["      }", "    },"]
    locales = ["en", "de", "ja", "ko", "zh-Hans", "zh-Hant"]
    lines = [f'    "{key}": {{', '      "extractionState": "manual",', '      "localizations": {']
    for index, locale in enumerate(locales):
        lines += [
            f'        "{locale}": {{',
            '          "stringUnit": {',
            '            "state": "translated",',
            f'            "value": {json.dumps(values[locale], ensure_ascii=False)}',
            "          }",
            "        }," if index < len(locales) - 1 else "        }",
        ]
    return lines + ["      }", "    },"]


def load(group):
    spec = GROUPS[group]
    text = Path(spec["path"]).read_text(encoding="utf-8")
    catalog_object = json.loads(text)
    if spec["form"] in ("canonical", "new"):
        assert canonical(catalog_object) == text, "not in Xcode's canonical form"
        keys = list(catalog_object["strings"])
        assert keys == sorted(keys), "keys not sorted"
    return text, catalog_object


def insert(group):
    spec = GROUPS[group]
    path = Path(spec["path"])
    strings = STRINGS[group]
    for key, values in strings.items():
        assert sorted(values) == sorted(LOCALES), key
    if spec["form"] == "new":
        assert not path.exists(), path
        setting = json.loads(SETTING.read_text(encoding="utf-8"))
        catalog_object = {
            "sourceLanguage": setting["sourceLanguage"],
            "strings": {key: entry_object(key, strings[key]) for key in sorted(strings)},
            "version": setting["version"],
        }
        path.parent.mkdir(parents=True, exist_ok=True)
        path.write_text(canonical(catalog_object), encoding="utf-8")
    elif spec["form"] == "canonical":
        text, catalog_object = load(group)
        old = catalog_object["strings"]
        assert not set(strings) & set(old), "a key is already there"
        merged = {}
        for key in old:
            if key == spec["anchor"]:
                for new_key in sorted(strings):
                    merged[new_key] = entry_object(new_key, strings[new_key])
            merged[key] = old[key]
        catalog_object["strings"] = merged
        path.write_text(canonical(catalog_object), encoding="utf-8")
    else:
        text, catalog_object = load(group)
        assert not set(strings) & set(catalog_object["strings"]), "a key is already there"
        lines = text.split("\n")
        separator = " : " if spec["form"] == "compact" else ": "
        anchor = f'    "{spec["anchor"]}"{separator}{{'
        assert lines.count(anchor) == 1, anchor
        new_lines = []
        for key, values in strings.items():
            new_lines += entry_lines(spec["form"], key, values)
        at = lines.index(anchor)
        lines[at:at] = new_lines
        path.write_text("\n".join(lines), encoding="utf-8")
    check(group)

    keys = list(catalog_object["strings"])

        expected = sorted(expected)
    at = keys.index(expected[0])
    assert keys[at:at + len(expected)] == expected, "the keys are not together in order"
    for key in expected:
        assert catalog_object["strings"][key] == entry_object(key, STRINGS[group][key]), key
    print("ok")
```

```swift
///
/// Confirm sends the change to the site and closes the sheet once the site took it; a refusal or a failure keeps it
/// open with the alert on the control that sent it. An add or a remove takes the page the site answered with into

@Reducer
public struct MyTagEditorFeature: Sendable {
    public enum Mode: Equatable, Sendable {
        case add
        /// The tag as it was, with the edit API's credentials of the page that opened it.
        case edit(MyTag, apiuid: Int, apikey: String)
    }

    /// A control that sends a request: confirm, or Remove from My Tags.
    public enum Control: Equatable, Sendable {
        case confirm
        case remove
    }

    @CasePathable
    public enum Delegate: Equatable, Sendable {
        case saved
        case removed
    }

    @ObservableState

        @Shared(.user) public var user: User
        @SharedReader(.setting) public var setting: Setting

        public var name: String
        public var kind: MyTagKind
        public var color: SiteColor?
        public var weight: Int

        case alert(PresentationAction<Never>)
        case closeTapped
        case confirmTapped

        case updateDone(MyTag, Result<MyTagEditReply, AppError>)
```

```swift
/// The field shows `color` as `#RRGGBB`, or nothing with `prompt` (else `#RRGGBB`) when there is none. It takes six
/// hex digits with or without `#`, in any case: the well follows as soon as the text is a valid code, and submit
/// writes the code back in upper case; submitting an empty field clears the colour, and an invalid code returns to
/// the last valid one. Picking in the well (`well`, no opacity: the site keeps `#rrggbb`) writes its colour.
public struct HexColorField: View {
    @ScaledMetric(relativeTo: .body) private var fieldWidth = 120.0
    @State private var text = ""
    @Binding private var color: SiteColor?
    @Binding private var well: Color
    private let title: LocalizedStringResource
    private let prompt: String?

    public init(_ title: LocalizedStringResource, color: Binding<SiteColor?>, well: Binding<Color>, prompt: String?) {
        self.title = title
        _color = color
        _well = well
        self.prompt = prompt
    }

    public var body: some View {
        LabeledContent {
            HStack(spacing: 12) {
                TextField(text: $text, prompt: promptText) {
                    Text(.myTagsHexColor)
                }
                .multilineTextAlignment(.trailing)
                .monospacedDigit()
                .keyboardType(.asciiCapable)
                .textInputAutocapitalization(.never)
                .autocorrectionDisabled(true)
                .submitLabel(.done)
                .accessibilityLabel(.myTagsHexColor)
                .frame(width: fieldWidth)
                .onSubmit {
                    submit()
                }
                ColorPicker(selection: $well, supportsOpacity: false) {
                    Text(title)
                }
                .labelsHidden()
            }
        } label: {
            Text(title)
        }
        // The field follows a colour set elsewhere (the well, Use Default Color), and leaves a code being typed alone.
        .onChange(of: color, initial: true) { _, newValue in
            if SiteColor(hex: text) != newValue {
                text = newValue?.hex ?? ""
            }
        }
        .onChange(of: text) { _, newValue in
            if let typed = SiteColor(hex: newValue), typed != color {
                color = typed
            }
        }
    }

    private var promptText: Text {
        if let prompt {
            return Text(verbatim: prompt)
        }
        return Text(.myTagsHexPrompt)
    }

    private func submit() {
        let trimmed = text.trimmingCharacters(in: .whitespaces)
        if trimmed.isEmpty {
            color = nil
        } else if let typed = SiteColor(hex: trimmed) {
            color = typed
            text = typed.hex
        } else {
            text = color?.hex ?? ""
        }
    }
}
```

```swift
public import AppComponents
public import AppModels
import AppTools
public import ComposableArchitecture
import CookieClient
public import Foundation
public import MyTagEditor
import MyTagsClient
import Resources
public import SwiftUI

/// The native My Tags page (port of PR 462 item 10, design approved 2026-11-22).
///
/// It reads every tag set when presented, on Retry, when the site's page in the web view closes and after a sign-in,
/// and the chosen set's page when the picker changes it or on pull to refresh. Enabled and the default colour go to
/// the site as they change, the colour after a second without another change, so a drag in the system picker sends
/// one request; a refusal or a failure puts the set back as the site has it and shows the alert. A row opens Edit
/// Tag, Add Tag the same form empty, and a swipe removes the tag at once. Every write takes the page the site
/// answered with into the cached My Tags. The edit API's credentials of the last page read stay in this state and are
/// never persisted.
@Reducer
public struct MyTagsFeature: Sendable {
    @Reducer
    public enum Destination {
        case editor(MyTagEditorFeature)
        /// The site's own My Tags page, the fallback when this page cannot read it.
        @ReducerCaseIgnored
        case browser(URL)
    }

    /// A control whose request failed, so the alert shows on it.
    public enum AlertSource: Equatable, Sendable {
        case enabled
        case defaultColor
        case tag(Int)
    }

    public enum Delegate: Equatable, Sendable {
        case pushLogin
    }

    private enum CancelID {
        case fetch
        case observeLogin
        case updateTagSet
    }

    @ObservableState
    public struct State: Equatable, Sendable {
        @Shared(.user) public var user: User
        @SharedReader(.setting) public var setting: Setting
        @Presents public var destination: Destination.State?
        @Presents public var alert: AppAlertState<Never>?
        @Presents public var toast: AppAlertState<Never>?
        public var alertSource: AlertSource?
        public var loadingState: LoadingState = .idle
        /// The edit API's credentials of the last page read; in memory only.
        public var apiuid: Int?
        public var apikey: String?
        /// The tag set as the site last had it while an Enabled or colour change is on its way.
        public var confirmedTagSet: MyTagSet?

        public init() {}

        public var selectedTagSet: MyTagSet? { user.myTags?.selectedTagSet }
        public var selectedTagSetID: Int { selectedTagSet?.id ?? 1 }
        public var isTagSetEnabled: Bool { selectedTagSet?.isEnabled ?? false }
        public var tagSetColor: SiteColor? { selectedTagSet?.color }
        public var tagSetWellColor: Color { selectedTagSet?.color?.color ?? .clear }
        /// The site's page the browser sheet shows; get-only, so the sheet binds through `browserChanged`.
        public var browserURL: URL? { destination?.browser }
    }

    public enum Action {
        case destination(PresentationAction<Destination.Action>)
        case alert(PresentationAction<Never>)
        case toast(PresentationAction<Never>)
        case delegate(Delegate)
        /// Sent when the page is presented.
        case onPresented
        case loginSucceeded
        case fetchMyTags
        case fetchMyTagsDone(Result<MyTagsResponse, AppError>)
        case tagSetSelected(Int)
        case refresh
        case fetchTagSetDone(Result<MyTagsPage, AppError>)
        case enabledChanged(Bool)
        case defaultColorChanged(SiteColor?)
        case defaultWellColorChanged(Color)
        case updateTagSetDone(AlertSource, Result<MyTagsPage, AppError>)
        case addTagTapped
        case tagTapped(MyTag)
        case removeSwiped(MyTag)
        case removeTagDone(tagID: Int, result: Result<MyTagsPage, AppError>)
        case openInBrowserTapped
        case browserChanged(URL?)
    }

    @Dependency(\.continuousClock) private var clock
    @Dependency(\.cookieClient) private var cookieClient
    @Dependency(\.myTagsClient) private var myTagsClient

    public init() {}

    public var body: some Reducer<State, Action> {
        Reduce { state, action in
            switch action {
            case .alert, .toast, .delegate:
                return .none

            case .onPresented:
                // Subscribe before returning the effect so a login during presentation is not missed.
                let changes = cookieClient.cookiesDidChange()
                let initiallyLoggedIn = cookieClient.didLogin
                let observeLogin: Effect<Action> = .run { send in
                    var wasLoggedIn = initiallyLoggedIn
                    for await _ in changes {
                        let isLoggedIn = cookieClient.didLogin
                        if !wasLoggedIn && isLoggedIn {
                            await send(.loginSucceeded)
                        }
                        wasLoggedIn = isLoggedIn
                    }
                }
                .cancellable(id: CancelID.observeLogin, cancelInFlight: true)
                guard initiallyLoggedIn else { return observeLogin }
                return .merge(observeLogin, .send(.fetchMyTags))

            case .loginSucceeded:
                return .send(.fetchMyTags)

            case .fetchMyTags:
                state.loadingState = .loading
                let host = state.setting.galleryHost
                return .run { send in
                    let result = await Result.Swift::init { () async throws(AppError) in
                        try await myTagsClient.fetchMyTags(host)
                    }
                    await send(.fetchMyTagsDone(result))
                }
                .cancellable(id: CancelID.fetch, cancelInFlight: true)

            case .fetchMyTagsDone(.success(let response)):
                state.loadingState = .idle
                state.apiuid = response.apiuid
                state.apikey = response.apikey
                state.$user.withLock({ $0.myTags = response.myTags.selecting($0.myTags?.selectedTagSetID) })
                return .none

            case .fetchMyTagsDone(.failure(let error)), .fetchTagSetDone(.failure(let error)):
                state.loadingState = .failed(error)
                return .none

            // The choice is kept with the cache, so Add to My Tags adds to the set last chosen here.
            case .tagSetSelected(let tagSetID):
                guard tagSetID != state.selectedTagSetID else { return .none }
                state.$user.withLock({ $0.myTags = $0.myTags?.selecting(tagSetID) })
                state.loadingState = .loading
                return fetchTagSet(state)

            case .refresh:
                return fetchTagSet(state)

            case .fetchTagSetDone(.success(let page)):
                state.loadingState = .idle
                take(page, into: &state)
                return .none

            case .enabledChanged(let isEnabled):
                return changeTagSet(&state, source: .enabled, waits: false) { $0.isEnabled = isEnabled }

            case .defaultColorChanged(let color):
                return changeTagSet(&state, source: .defaultColor, waits: true) { $0.color = color }

            case .defaultWellColorChanged(let color):
                return changeTagSet(&state, source: .defaultColor, waits: true) { $0.color = SiteColor(color) }

            // The page the site answered with is the set as it now is, so it also puts back a change it refused.
            case .updateTagSetDone(let source, .success(let page)):
                state.confirmedTagSet = nil
                take(page, into: &state)
                guard let message = page.message else { return .none }
                return presentAlert(&state, source, title: .RLocalizable.myTagsSaveFailed, message: message)

            case .updateTagSetDone(let source, .failure(let error)):
                if let confirmed = state.confirmedTagSet {
                    state.$user.withLock({ $0.myTags = $0.myTags?.updating(confirmed) })
                }
                state.confirmedTagSet = nil
                return presentAlert(&state, source, title: .RLocalizable.myTagsSaveFailed, message: error.alertText)

            case .addTagTapped:
                guard let tagSet = state.selectedTagSet else { return .none }
                state.destination = .editor(MyTagEditorFeature.State(tagSet: tagSet))
                return .none

            case .tagTapped(let tag):
                guard let tagSet = state.selectedTagSet, let apiuid = state.apiuid, let apikey = state.apikey
                else { return .none }
                state.destination = .editor(
                    MyTagEditorFeature.State(editing: tag, tagSet: tagSet, apiuid: apiuid, apikey: apikey)
                )
                return .none

            case .removeSwiped(let tag):
                guard let tagSetID = state.selectedTagSet?.id else { return .none }
                let host = state.setting.galleryHost
                return .run { send in
                    let result = await Result.Swift::init { () async throws(AppError) in
                        try await myTagsClient.removeTags(host, [tag.id], tagSetID)
                    }
                    // The row leaves with the list's animation.
                    await send(.removeTagDone(tagID: tag.id, result: result), animation: .default)
                }

            case .removeTagDone(let tagID, .success(let page)):
                take(page, into: &state)
                guard let message = page.message else { return .none }
                return presentAlert(&state, .tag(tagID), title: .RLocalizable.myTagsRemoveFailed, message: message)

            case .removeTagDone(let tagID, .failure(let error)):
                return presentAlert(
                    &state, .tag(tagID), title: .RLocalizable.myTagsRemoveFailed, message: error.alertText
                )

            case .destination(.presented(.editor(.delegate(.removed)))):
                state.toast = .success(caption: .RLocalizable.myTagsRemoved)
                return .none

            case .openInBrowserTapped:
                state.destination = .browser(Defaults.URL.myTags(host: state.setting.galleryHost))
                return .none

            // The browser is a state-only destination, so closing it arrives here as nil rather than as a dismiss.
            // Closing the site's page reads My Tags again: the tags may have changed there.
            case .browserChanged(let url):
                guard url == nil, state.destination?.is(\.browser) == true else { return .none }
                state.destination = nil
                return .send(.fetchMyTags)

            case .destination:
                return .none
            }
        }
        .ifLet(\.$destination, action: \.destination)
        .ifLet(\.$alert, action: \.alert)
        .ifLet(\.$toast, action: \.toast)
    }

    private func fetchTagSet(_ state: State) -> Effect<Action> {
        let host = state.setting.galleryHost
        let tagSetID = state.selectedTagSetID
        return .run { send in
            let result = await Result.Swift::init { () async throws(AppError) in
                try await myTagsClient.fetchTagSet(host, tagSetID)
            }
            await send(.fetchTagSetDone(result))
        }
        .cancellable(id: CancelID.fetch, cancelInFlight: true)
    }

    /// Takes a page the site served into the cache, with its credentials.
    private func take(_ page: MyTagsPage, into state: inout State) {
        state.apiuid = page.apiuid
        state.apikey = page.apikey
        state.$user.withLock({ $0.myTags = $0.myTags?.applying(page) })
    }

    /// Puts the change into the cache at once, then sends the set as it now is; with `waits`, a second without another
    /// change first.
    private func changeTagSet(
        _ state: inout State,
        source: AlertSource,
        waits: Bool,
        change: (inout MyTagSet) -> Void
    ) -> Effect<Action> {
        guard var tagSet = state.selectedTagSet else { return .none }

                try await myTagsClient.updateTagSet(host, tagSet.id, tagSet.isEnabled, tagSet.color)
```
